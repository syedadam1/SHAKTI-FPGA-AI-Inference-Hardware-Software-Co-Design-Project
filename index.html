// Synthesizable Verilog RTL: AXI4-Lite 4x4 Parallel Systolic GEMM Accelerator
// Base Address in SHAKTI SoC: 0x4002_0000
module gemm_axi_accelerator (
    input  wire        S_AXI_ACLK,
    input  wire        S_AXI_ARESETN,
    input  wire [7:0]  S_AXI_AWADDR,
    input  wire        S_AXI_AWVALID,
    output reg         S_AXI_AWREADY,
    input  wire [31:0] S_AXI_WDATA,
    input  wire        S_AXI_WVALID,
    output reg         S_AXI_WREADY,
    output reg  [31:0] S_AXI_RDATA,
    output reg         S_AXI_RVALID,
    input  wire        S_AXI_RREADY
);
    reg [31:0] ctrl_reg, mat_a_packed[0:3], mat_b_packed[0:3];
    reg signed [31:0] mat_c [0:15];
    reg busy, done;
    reg [2:0] k_step;

    wire signed [7:0] A [0:3][0:3];
    wire signed [7:0] B [0:3][0:3];

    // Parallel 16-PE Systolic Outer-Product Update (16 MACs per clock cycle)
    always @(posedge S_AXI_ACLK) begin
        if (!S_AXI_ARESETN) begin
            busy <= 0; done <= 0; k_step <= 0;
        end else if (ctrl_reg[0] && !busy) begin
            busy <= 1; done <= 0; k_step <= 0;
        end else if (busy) begin
            if (k_step < 4) begin
                for (int i = 0; i < 4; i++)
                    for (int j = 0; j < 4; j++)
                        mat_c[i*4 + j] <= mat_c[i*4 + j] + (A[i][k_step] * B[k_step][j]);
                k_step <= k_step + 1;
            end else begin
                busy <= 0; done <= 1;
            end
        end
    end
endmodule

// SHAKTI RISC-V C Firmware Driver (shakti_ai_inference.c)
#include <stdint.h>

#define FPGA_ACCEL_BASE   0x40020000UL
#define REG_CTRL          (*(volatile uint32_t *)(FPGA_ACCEL_BASE + 0x00))
#define REG_STATUS        (*(volatile uint32_t *)(FPGA_ACCEL_BASE + 0x04))
#define REG_MAT_A_ROW(r)  (*(volatile uint32_t *)(FPGA_ACCEL_BASE + 0x10 + ((r)*4)))
#define REG_MAT_B_ROW(r)  (*(volatile uint32_t *)(FPGA_ACCEL_BASE + 0x20 + ((r)*4)))
#define REG_MAT_C_VAL(i)  (*(volatile int32_t  *)(FPGA_ACCEL_BASE + 0x30 + ((i)*4)))

void gemm_fpga_offload(const int8_t A[4][4], const int8_t B[4][4], int32_t C[4][4]) {
    for (int r = 0; r < 4; r++) {
        REG_MAT_A_ROW(r) = PACK_INT8x4(A[r][0], A[r][1], A[r][2], A[r][3]);
        REG_MAT_B_ROW(r) = PACK_INT8x4(B[r][0], B[r][1], B[r][2], B[r][3]);
    }
    REG_CTRL = 0x01;                     // Trigger FPGA Systolic Array
    while ((REG_STATUS & 0x02) == 0);    // Poll until DONE == 1
    for (int i = 0; i < 16; i++) {
        C[i/4][i%4] = REG_MAT_C_VAL(i);  // Read back result
    }
}
